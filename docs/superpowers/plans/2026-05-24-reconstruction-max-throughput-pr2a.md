```python
"""Reconstruction terrain upper-bound capacity (output-only; never solver input)."""

from __future__ import annotations

from decimal import Decimal
from typing import Any

from django_apps.asteroid_lab.reconstruction.result import ReconstructionResult
from django_apps.game_data.services.mining_extraction_rules import (
    effective_mini_units,
    get_active_rule,
    max_output_per_miner,
)


def decimal_str(value: Decimal) -> str:
    return format(value.quantize(Decimal("0.0001")), "f")


def build_reconstruction_capacity_summary(
    *,
    recon: ReconstructionResult,
    resource_kind: str,
) -> dict[str, Any]:
    rule = get_active_rule(resource_kind)
    platform_count = len(recon.confirmed_cells)
    per_miner = max_output_per_miner(rule)
    total = per_miner * Decimal(platform_count)
    max_mini = effective_mini_units(int(rule.max_extension_count))
    return {
        "resource_kind": resource_kind,
        "capacity_upper_bound_platform_count": platform_count,
        "mini_unit_output_per_min": decimal_str(rule.mini_unit_output_per_min),
        "max_mini_units_per_miner": max_mini,
        "max_output_per_miner": decimal_str(per_miner),
        "max_throughput_per_min": decimal_str(total),
        "output_unit": rule.output_unit,
        "source_kind": rule.source_kind,
        "authority": "MiningExtractionRule",
    }


def build_reconstruction_capacity_envelope(*, recon: ReconstructionResult) -> dict[str, Any]:
    return {
        "capacity_basis": "terrain_upper_bound",
        "by_resource": {
            "shape": build_reconstruction_capacity_summary(recon=recon, resource_kind="shape"),
            "fluid": build_reconstruction_capacity_summary(recon=recon, resource_kind="fluid"),
        },
    }


def build_reconstruction_observability(*, recon: ReconstructionResult) -> dict[str, Any]:
    obs: dict[str, Any] = {
        "cell_count": len(recon.cells),
        "confirmed_cell_count": len(recon.confirmed_cells),
        "ambiguous_cell_count": len(recon.ambiguous_cells),
        "external_void_cell_count": len(recon.external_void_cells),
        "quality_tier": str(recon.quality_tier),
        "confidence_score": decimal_str(Decimal(str(recon.confidence_score))),
    }
    summary = dict(recon.summary_json or {})
    for key in ("inferred_shell_cell_count",):
        if key in summary:
            obs[key] = summary[key]
    return obs


__all__ = [
    "build_reconstruction_capacity_envelope",
    "build_reconstruction_capacity_summary",
    "build_reconstruction_observability",
    "decimal_str",
]
```

```python
_QUALITY_TIER_SHORT = {
    "CONFIDENT_RECONSTRUCTION": "HIGH",
    "PROVISIONAL_RECONSTRUCTION": "MED",
}


def _quality_tier_short(tier: str) -> str:
    return _QUALITY_TIER_SHORT.get(tier, tier[:4].upper() if tier else "—")


def _section_reconstruction(obs: dict[str, Any] | None) -> dict[str, Any]:
    if not obs:
        return {k: _PLACEHOLDER for k in (
            "cell_count", "confirmed_cell_count", "ambiguous_cell_count",
            "external_void_cell_count", "quality_tier", "confidence_score", "quality_tier_short",
        )}
    tier = str(obs.get("quality_tier", _PLACEHOLDER))
    return {
        "cell_count": obs.get("cell_count", _PLACEHOLDER),
        "confirmed_cell_count": obs.get("confirmed_cell_count", _PLACEHOLDER),
        "ambiguous_cell_count": obs.get("ambiguous_cell_count", _PLACEHOLDER),
        "external_void_cell_count": obs.get("external_void_cell_count", _PLACEHOLDER),
        "quality_tier": tier,
        "confidence_score": obs.get("confidence_score", _PLACEHOLDER),
        "quality_tier_short": _quality_tier_short(tier) if tier != _PLACEHOLDER else _PLACEHOLDER,
    }


def _section_capacity(cap: dict[str, Any] | None) -> dict[str, Any]:
    empty = {
        "shape_max_throughput_per_min": _PLACEHOLDER,
        "fluid_max_throughput_per_min": _PLACEHOLDER,
        "shape_output_unit": _PLACEHOLDER,
        "fluid_output_unit": _PLACEHOLDER,
        "reconstruction_max_throughput_per_min": _PLACEHOLDER,
        "platform_upper_bound": _PLACEHOLDER,
        "capacity_basis": _PLACEHOLDER,
        "extraction_rule_source": _PLACEHOLDER,
    }
    if not cap:
        return empty
    by = dict(cap.get("by_resource") or {})
    shape = dict(by.get("shape") or {})
    fluid = dict(by.get("fluid") or {})
    platform = shape.get("capacity_upper_bound_platform_count", _PLACEHOLDER)
    return {
        "shape_max_throughput_per_min": shape.get("max_throughput_per_min", _PLACEHOLDER),
        "fluid_max_throughput_per_min": fluid.get("max_throughput_per_min", _PLACEHOLDER),
        "shape_output_unit": shape.get("output_unit", _PLACEHOLDER),
        "fluid_output_unit": fluid.get("output_unit", _PLACEHOLDER),
        "reconstruction_max_throughput_per_min": shape.get("max_throughput_per_min", _PLACEHOLDER),
        "platform_upper_bound": platform,
        "capacity_basis": cap.get("capacity_basis", _PLACEHOLDER),
        "extraction_rule_source": shape.get("source_kind", _PLACEHOLDER),
    }


def _section_rttp(solver_summary: dict[str, Any]) -> dict[str, Any]:
    order = list(solver_summary.get("commit_order") or [])
    preview = order[0] if order else _PLACEHOLDER
    if len(order) > 1:
        preview = f"{order[0]} (+{len(order) - 1})"
    actual = solver_summary.get("actual_committed_output_per_min")
    return {
        "confirmed_count": solver_summary.get("confirmed_count", _PLACEHOLDER),
        "validation_passed": bool(solver_summary.get("validation_passed")),
        "actual_committed_output_per_min": actual,
        "actual_output_status": "available" if actual is not None else "pending_pr_2b",
        "candidate_count": solver_summary.get("normal_candidate_count", _PLACEHOLDER),
        "commit_order_preview": preview,
    }
```

```javascript
function formatThroughputDisplay(value, unit) {
  const dash = "—";
  if (value == null || value === "" || value === dash) return dash;
  const n = Number.parseFloat(String(value));
  const num = Number.isFinite(n) ? String(Math.round(n)) : String(value);
  return unit ? num + " " + unit : num;
}

function updateLabStatCards(run) {
  const dash = "—";
  const cap = run && run.capacity ? run.capacity : {};
  const rec = run && run.reconstruction ? run.reconstruction : {};
  const rttp = run && run.rttp ? run.rttp : {};
  const set = (id, text) => {
    const el = document.getElementById(id);
    if (el) el.textContent = text != null ? String(text) : dash;
  };
  set(
    "lab-card-theoretical-max-value",
    formatThroughputDisplay(cap.shape_max_throughput_per_min, cap.shape_output_unit),
  );
  set(
    "lab-card-resource-capacity-value",
    cap.shape_max_throughput_per_min !== dash && cap.fluid_max_throughput_per_min !== dash
      ? shapezUiT("Belt") + " " + formatThroughputDisplay(cap.shape_max_throughput_per_min, "") +
          " / " + shapezUiT("Fluid") + " " + formatThroughputDisplay(cap.fluid_max_throughput_per_min, "")
      : dash,
  );
  set(
    "lab-card-footprint-value",
    rec.confirmed_cell_count !== dash && rec.cell_count !== dash
      ? rec.confirmed_cell_count + " / " + rec.cell_count
      : dash,
  );
  set(
    "lab-card-reconstruction-quality-value",
    rec.quality_tier_short !== dash
      ? rec.quality_tier_short + " · " + (rec.confidence_score != null ? rec.confidence_score : dash)
      : dash,
  );
  const placements =
    rttp.confirmed_count !== dash && rttp.confirmed_count !== undefined
      ? String(rttp.confirmed_count) + " " + shapezUiT("placement(s)")
      : dash;
  set("lab-card-rttp-committed-value", placements);
  const pendingEl = document.getElementById("lab-card-rttp-committed-sub");
  if (pendingEl) {
    pendingEl.textContent =
      rttp.actual_output_status === "pending_pr_2b"
        ? shapezUiT("actual output pending")
        : dash;
  }
}
```
